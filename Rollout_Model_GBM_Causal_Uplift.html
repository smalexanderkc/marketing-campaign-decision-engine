def lift_at_pct(y_true, scores, pct):

    df = (
        pd.DataFrame({
            "target": y_true,
            "score": scores
        })
        .sort_values("score", ascending=False)
    )

    n = max(1, int(len(df) * pct))

    top_rate = df.head(n)["target"].mean()
    overall_rate = df["target"].mean()

    return top_rate / overall_rate

def calc_metrics(cm):

    tn, fp, fn, tp = cm.ravel()

    accuracy = (tp + tn) / cm.sum()

    recall = (
        tp / (tp + fn)
        if (tp + fn) > 0
        else np.nan
    )

    precision = (
        tp / (tp + fp)
        if (tp + fp) > 0
        else np.nan
    )

    specificity = (
        tn / (tn + fp)
        if (tn + fp) > 0
        else np.nan
    )

    f1 = (
        2 * precision * recall /
        (precision + recall)
        if (precision + recall) > 0
        else np.nan
    )

    return {
        "TP": tp,
        "TN": tn,
        "FP": fp,
        "FN": fn,
        "Accuracy": accuracy,
        "Recall": recall,
        "Precision": precision,
        "Specificity": specificity,
        "F1": f1
    }
